    if type(writefile) ~= "function" then return end

        smallServer and "1" or "0",

        selected.Candy and "1" or "0",

        writefile(

local function queueAutoExecute()
    if not autoExecute or queueRegistered then return end
    local queue = queue_on_teleport
        or (syn and syn.queue_on_teleport)
        or (fluxus and fluxus.queue_on_teleport)
    if type(queue) ~= "function"
        or type(isfile) ~= "function"
        or type(readfile) ~= "function"
        or not isfile(CORE_FILE) then
        warn("Auto Execute requires queue_on_teleport and local file access.")
        return
    end
    local launcher = [[
        if type(readfile) ~= "function"
            or type(isfile) ~= "function" then
            return
        end
        local cfg = "galegay_farm_settings.txt"
        local core = "galegay_farm_core.lua"
        local stopped = "galegay_farm_stopped.flag"
        if isfile(stopped)
            and readfile(stopped) == "1" then
            return
        end
        if not isfile(cfg) or not isfile(core) then
            return
        end
        local ok, data = pcall(function()
            return readfile(cfg)
        end)
        if not ok
            or type(data) ~= "string"
            or string.sub(data, 1, 2) ~= "1|" then
            return
        end
        local okCore, source = pcall(function()
            return readfile(core)
        end)
        if okCore
            and type(source) == "string"
            and #source > 100 then
            if type(writefile) == "function" then
                pcall(function()
                    writefile(
                        "galegay_farm_hopped.flag",
                        "1"
                    )
                end)
            end
            local run, err = loadstring(source)
            if run then
                local env = _G
                if type(getgenv) == "function" then
                    local okEnv, result = pcall(getgenv)
                    if okEnv and type(result) == "table" then
                        env = result
                    end
                end
                env.__KSTG_FARM_HOP_JOB = game.JobId
                run()
            else
                warn(err)
            end
        end
    ]]
    local ok, err = pcall(function()
        queue(launcher)
    end)
    if ok then
        queueRegistered = true
    else
        warn("Auto Execute: " .. tostring(err))
    end
end

local function serverhop()
    if not alive or stopTriggered then return false end
    local candidates = {}
    local cursor = nil
    for page = 1, 3 do
        local url =
            "https://games.roblox.com/v1/games/"
            .. game.PlaceId
            .. "/servers/Public?sortOrder=Asc&limit=100"
        if cursor and cursor ~= "" then
            url = url
                .. "&cursor="
                .. HttpService:UrlEncode(cursor)
        end
        local ok, response = pcall(function()
            return game:HttpGetAsync(url)
        end)
        if not ok then
            statusText.Text =
                "SERVERHOP unavailable: HTTP request failed."
            return false
        end
        local decodedOk, body = pcall(function()
            return HttpService:JSONDecode(response)
        end)
        if not decodedOk
            or type(body) ~= "table" then
            statusText.Text =
                "SERVERHOP unavailable: invalid server list."
            return false
        end
        for _, server in ipairs(body.data or {}) do
            if type(server) == "table"
                and type(server.playing) == "number"
                and type(server.maxPlayers) == "number"
                and type(server.id) == "string"
                and server.id ~= game.JobId
                and server.playing < server.maxPlayers
                and (
                    not smallServer
                    or server.playing <= 5
                ) then
                table.insert(candidates, server)
            end
        end
        if #candidates > 0 then
            break
        end
        cursor = body.nextPageCursor
        if not cursor then break end
    end
    if #candidates == 0 then
        statusText.Text = smallServer
            and "No small servers found."
            or "No available servers found."
        return false
    end
    table.sort(candidates, function(a, b)
        return a.playing < b.playing
    end)
    local poolSize = smallServer
        and math.min(5, #candidates)
        or #candidates
    local target = candidates[
        math.random(1, poolSize)
    ]
    if not alive or stopTriggered then return false end
    saveSettings()
    queueAutoExecute()
    statusText.Text =
        "SERVERHOP: " .. target.playing
        .. "/" .. target.maxPlayers .. " players"
    local ok, err = pcall(function()
        TeleportService:TeleportToPlaceInstance(
            game.PlaceId, target.id, player
        )
    end)
    if not ok then
        statusText.Text =
            "Teleport error: " .. tostring(err)
    end
    return ok
end

local function executeSelected()
    if not alive or stopTriggered then return end
    if smallServer
        and #Players:GetPlayers() > 5 then
        statusText.Text =
            "Small Server ON: use SERVERHOP."
        return
    end
    local requested, started = 0, 0
    for _, mode in ipairs({
        "Snow", "Candy", "Slapple"
    }) do
        if selected[mode] then
            requested += 1
            local run = farmScripts[mode]
            if type(run) == "function"
                and not running[mode] then
                running[mode] = true
                started += 1
                workers[mode] = task.spawn(function()
                    local ok, err = pcall(run)
                    running[mode] = false
                    workers[mode] = nil
                    if not ok then
                        warn(
                            "GALEGAY Farm "
                            .. mode .. ": "
                            .. tostring(err)
                        )
                    end
                end)
            end
        end
    end
    if requested == 0 then
        statusText.Text =
            "Select at least one farm before executing."
    elseif started == 0 then
        statusText.Text =
            "Selected scripts are already running."
    elseif started < requested then
        statusText.Text =
            "Started " .. started .. " script(s)."
    else
        statusText.Text =
            "Starting " .. started .. " script(s)."
    end
end

    end)

        warn("Could not save hub settings: " .. tostring(err))
    end
end
